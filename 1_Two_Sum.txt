class Solution {
    public int[] twoSum(int[] nums, int target) {
        int[] output = new int[2];

        for(int i=0;i<nums.length;i++){
            for(int n=i+1;n<nums.length;n++){
                if(nums[i] + nums[n] == target){
                    output[0] = i;
                    output[1] = n;
                    return output;
                }
            }
        }
        return output;
    }
}
