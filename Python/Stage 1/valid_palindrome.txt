class Solution(object):
    def isPalindrome(self, s):
        """
        :type s: str
        :rtype: bool
        """
        s = s.lower()
        cleaned = []

        for char in s:
            if char.isalnum():
                cleaned.append(char)

        cleaned = "".join(cleaned)

        left = 0
        right = len(cleaned) - 1

        while left < right:
            if cleaned[left] != cleaned[right]:
                return False
            else:
                left += 1
                right -= 1
        
        return True
